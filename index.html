  <script>

    /* ------------------------------------------------------------------

       1. SCROLL-DRIVEN CELESTIAL ORB & SKY BACKGROUND INTERPOLATION

       ------------------------------------------------------------------ */

    const bgSunrise = document.getElementById('bg-sunrise');

    const bgDay = document.getElementById('bg-day');

    const bgSunset = document.getElementById('bg-sunset');

    const bgTwilight = document.getElementById('bg-twilight');

    const bgNight = document.getElementById('bg-night');

    const starfield = document.getElementById('starfield');

 

    const celestialOrb = document.getElementById('celestial-orb');

 

    window.addEventListener('scroll', updateScrollEngine);

 

    function updateScrollEngine() {

      const scrollTop = window.scrollY;

      const maxScroll = document.documentElement.scrollHeight - window.innerHeight;

      const scrollPct = Math.min(Math.max(scrollTop / maxScroll, 0), 1);

 

      // Interpolate Background Sky Opacities

      if (scrollPct < 0.25) {

        // Sunrise -> Day

        const p = scrollPct / 0.25;

        bgSunrise.style.opacity = 1 - p;

        bgDay.style.opacity = p;

        bgSunset.style.opacity = 0;

        bgTwilight.style.opacity = 0;

        bgNight.style.opacity = 0;

        starfield.style.opacity = 0;

      } else if (scrollPct < 0.50) {

        // Day -> Sunset

        const p = (scrollPct - 0.25) / 0.25;

        bgSunrise.style.opacity = 0;

        bgDay.style.opacity = 1 - p;

        bgSunset.style.opacity = p;

        bgTwilight.style.opacity = 0;

        bgNight.style.opacity = 0;

        starfield.style.opacity = 0;

      } else if (scrollPct < 0.75) {

        // Sunset -> Twilight

        const p = (scrollPct - 0.50) / 0.25;

        bgSunrise.style.opacity = 0;

        bgDay.style.opacity = 0;

        bgSunset.style.opacity = 1 - p;

        bgTwilight.style.opacity = p;

        bgNight.style.opacity = 0;

        starfield.style.opacity = p * 0.5;

      } else {

        // Twilight -> Night (Full Moon)

        const p = (scrollPct - 0.75) / 0.25;

        bgSunrise.style.opacity = 0;

        bgDay.style.opacity = 0;

        bgSunset.style.opacity = 0;

        bgTwilight.style.opacity = 1 - p;

        bgNight.style.opacity = p;

        starfield.style.opacity = 0.5 + (p * 0.5
